The Dual-Stage Pipeline

Traditional CLIP models often suffer from the compositional binding problem—failing to associate specific attributes with nouns (e.g. matching "red cup and blue plate").

MIRAGE solves this by decoupling the search into a two-stage process:

  • 1

    VLM Dense Captioning (Coarse Filter)

    Ingests images and uses a Vision-Language Model to write semantic descriptors. Standard dense retrieval isolates candidates with correct grammar/color binding.

  • 2

    CLIP Visual Grounding (Reranker)

    Scores the top-N candidates against the query vector in CLIP's joint-embedding space to compute exact pixel alignments, fusing both scores for final ranking.

Quick Start Guide

1. Clone & Install Dependencies

Copy
git clone https://github.com/kanaad-lims/mirage.git cd mirage pip install -r requirements.txt

2. Configure Environment Keys

Create a .env file in the project root and add your Moondream VLM API Key:

MOONDREAM_API_KEY=your_api_key_here

3. Index Your Images

Place your images (e.g. .jpg, .png) inside the data/images/ folder and run the builder script. Features are cached automatically for skip/resume resilience.

python -m indexer.build_index

4. Search via Natural Language

Search your visual collection using the retriever CLI. The script stitches matching results into a horizontal preview grid.

python -m retriever.search a red coffee mug on a white desk

import os from dotenv import load_dotenv # Load variables from .env load_dotenv() MOONDREAM_API_KEY = os.getenv("MOONDREAM_API_KEY") # Models Specifications TEXT_EMBED_MODEL = "all-MiniLM-L6-v2" CLIP_MODEL = "openai/clip-vit-base-patch32" # General-purpose CLIP # Workspace Paths BASE_DIR = os.path.dirname(os.path.abspath(__file__)) IMAGES_DIR = os.path.join(BASE_DIR, "data", "images") CHROMA_DIR = os.path.join(BASE_DIR, "data", "chroma_db") COLLECTION_NAME = "mirage_image_collection" # Search parameters (Score Fusion Weights) TOP_N_CANDIDATES = 20 CAPTION_WEIGHT = 0.3 CLIP_WEIGHT = 0.7
import torch from transformers import CLIPProcessor, CLIPModel from sentence_transformers import SentenceTransformer from config import TEXT_EMBED_MODEL, CLIP_MODEL device = "cuda" if torch.cuda.is_available() else "cpu" # Initialize models in GPU/CPU _text_model = SentenceTransformer(TEXT_EMBED_MODEL, device=device) _clip_model = CLIPModel.from_pretrained(CLIP_MODEL).to(device) _clip_processor = CLIPProcessor.from_pretrained(CLIP_MODEL) def embed_caption(caption: str): """Encodes caption text using SentenceTransformer""" return _text_model.encode(caption).tolist() def embed_query_text(text: str): """Encodes query string into standard CLIP text-space""" inputs = _clip_processor(text=[text], return_tensors="pt", padding=True).to(device) with torch.no_grad(): feats = _clip_model.get_text_features(**inputs) feats = feats / feats.norm(dim=-1, keepdim=True) return feats.squeeze().tolist()
import ast import numpy as np from config import TOP_N_CANDIDATES, CAPTION_WEIGHT, CLIP_WEIGHT from indexer.embedder import embed_caption, embed_query_text def _cosine(a, b): a, b = np.array(a), np.array(b) return float(np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b))) class QueryEngine: def search_images(self, query: str, k: int = 5): # Stage 1: Coarse-filter candidates in ChromaDB results = self.store.query(embed_caption(query), n_results=TOP_N_CANDIDATES) query_text_emb = embed_query_text(query) candidates = [] # 2. Applying reranking on top_n images # cosine -> Image embeddings and CLIP text embeddings (not semantic query embedding) for i in range(len(results["ids"][0])): caption_score = 1 - (results["distances"][0][i] / 2) clip_image_emb = ast.literal_eval(results["metadatas"][0][i]["clip_embedding"]) clip_score = _cosine(query_text_emb, clip_image_emb) # Multi-modal score fusion final_score = (CAPTION_WEIGHT * caption_score) + (CLIP_WEIGHT * clip_score) candidates.append({ "image_path": results["metadatas"][0][i]["image_path"], "score": final_score }) candidates.sort(key=lambda x: x["score"], reverse=True) return candidates[:k]
